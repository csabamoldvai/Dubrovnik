# Example script: read-only device check.
# Run it with File > Run Script... (Esc stops it).
# Nothing here erases, programs or changes the board settings.

ver                 # Tiberias software version
config              # current board configuration
9f 2                # read the 2-byte JEDEC ID of the flash
05                  # status register 1
35                  # status register 2
b 0 20              # read 0x20 bytes from address 0
